SELECTION-SCREEN: BEGIN OF BLOCK b1 WITH FRAME TITLE t1.
  PARAMETERS: order    RADIOBUTTON GROUP doc USER-COMMAND radio DEFAULT 'X',
              packslip RADIOBUTTON GROUP doc,
              invoice  RADIOBUTTON GROUP doc.
SELECTION-SCREEN: END OF BLOCK b1.

SELECTION-SCREEN: BEGIN OF BLOCK b2 WITH FRAME TITLE t2.
  PARAMETERS: vbeln        LIKE vbak-vbeln MODIF ID sc1 MATCHCODE OBJECT vmva,
              pbeln        LIKE vbak-vbeln MODIF ID sc3 MATCHCODE OBJECT vmvl,
              ibeln        LIKE vbrk-vbeln MODIF ID sc2 MATCHCODE OBJECT vmcf,
              address1     LIKE tsp02-pjposname,
              address2     LIKE tsp02-pjposname,
              address3     LIKE tsp02-pjposname,
              address4     LIKE tsp02-pjposname,
              subject(250).
SELECTION-SCREEN: END OF BLOCK b2.

SELECTION-SCREEN: BEGIN OF BLOCK b3 WITH FRAME TITLE t3.
  PARAMETERS: xsave    AS CHECKBOX,
              saveonly AS CHECKBOX.
SELECTION-SCREEN: END OF BLOCK b3.

DATA: w_tim LIKE sy-uzeit.



AT SELECTION-SCREEN.

  IF order = 'X' AND vbeln = ''.
    MESSAGE e203(zsd).
  ELSEIF invoice = 'X' AND ibeln = '' .
    MESSAGE e204(zsd).
  ELSEIF packslip = 'X' AND pbeln = ''.
    MESSAGE e202(zsd).
  ENDIF.

  IF order = 'X'.
    SELECT SINGLE vbeln INTO vbak-vbeln FROM vbak
        WHERE vbeln = vbeln.
    IF sy-subrc <> 0.
      MESSAGE e205(zsd) WITH vbeln.
    ENDIF.
    CLEAR: pbeln, ibeln.
  ELSEIF invoice = 'X'.
    SELECT SINGLE vbeln kunag INTO (vbrk-vbeln, vbrk-kunag) FROM vbrk
        WHERE vbeln = ibeln.
    IF sy-subrc <> 0.
      MESSAGE e206(zsd) WITH ibeln.
    ENDIF.
    CLEAR: vbeln, pbeln.
  ELSEIF packslip = 'X'.
    SELECT SINGLE vbeln kunnr INTO (likp-vbeln, likp-kunag) FROM likp
      WHERE vbeln = pbeln.
    IF sy-subrc <> 0.
      MESSAGE e207(zsd) WITH pbeln.
    ENDIF.
    CLEAR: vbeln, ibeln.
  ENDIF.

  IF address1 = '' AND saveonly = ''.
    MESSAGE e209(zsd).
  ENDIF.

AT SELECTION-SCREEN OUTPUT.


INITIALIZATION.

  MOVE 'Details:' TO t2.
  MOVE 'Document type:' TO t1.
  MOVE 'Archiving' TO t3.

START-OF-SELECTION.
  w_tim = sy-uzeit.
  title = subject.
  IF xsave = 'X' OR saveonly = 'X'.
    PERFORM get_path.
  ENDIF.
  IF order = 'X'.
    PERFORM email_sales_order.
  ELSEIF invoice = 'X'.
    PERFORM email_invoice.
  ELSEIF packslip = 'X'.
    PERFORM email_packing_slip.
  ENDIF.

*&---------------------------------------------------------------------*
*&      Form  EMAIL_SALES_ORDER
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM email_sales_order .

* SEND OUTPUT TO SPOOL
  IF title = ''.
    title = 'SALES ORDER '.
    title+12(10) = vbeln.
    title+33(8) = w_tim.
  ELSE.
    CONCATENATE title w_tim INTO title.
  ENDIF.
  CLEAR vbak-auart.
  SELECT SINGLE auart kunnr FROM vbak
          INTO  (vbak-auart, vbak-kunnr)
          WHERE vbeln = vbeln.
  CLEAR name1.                      " 11989 - Find customer name
  SELECT SINGLE name1 FROM kna1
          INTO name1
          WHERE kunnr = vbak-kunnr.
  REFRESH bdcdata.
  PERFORM bdc_dynpro USING 'SAPMV45A'         '0102'.
  PERFORM bdc_field  USING 'VBAK-VBELN'        vbeln.
  PERFORM bdc_field  USING 'BDC_OKCODE'       'DRCK'.
  PERFORM bdc_dynpro USING 'SAPLVMSG'         '0200'.
  PERFORM bdc_field  USING 'BDC_CURSOR'       'NAST-KSCHL(01)'.
  PERFORM bdc_field  USING 'BDC_OKCODE'       'OPTS'.
  PERFORM bdc_dynpro USING 'SAPLVMSG'         '0100'.
  PERFORM bdc_field  USING 'NAST-DIMME'      ' '.
  PERFORM bdc_field  USING 'NAST-DELET'      'X'.
  PERFORM bdc_field  USING 'NAST-TDCOVTITLE'  title.
  PERFORM bdc_field  USING 'BDC_OKCODE'       'STAR'.
  PERFORM bdc_dynpro USING 'SAPLVMSG'         '0200'.
  PERFORM bdc_field  USING 'BDC_OKCODE'       'PRNT'.
  IF vbak-auart = 'AG'.
    PERFORM call_transaction_va23.
  ELSE.
    PERFORM call_transaction_va03.
  ENDIF.

  IF xsave = '' AND saveonly = ''.
    PERFORM email.
  ELSEIF xsave = 'X' AND saveonly = '' .
    PERFORM spool_convert_to_pdf.
    PERFORM email.
  ELSEIF saveonly = 'X'.
    PERFORM spool_convert_to_pdf.
  ENDIF.


ENDFORM.                    " EMAIL_SALES_ORDER

  packing_list-doc_type   = 'PDF'.
  packing_list-obj_name   = 'MAIL'.

  packing_list-obj_descr  = pdf_name.
  APPEND packing_list.
  REFRESH i_head.
  i_head = 'HF Doc.pdf'.
  APPEND i_head.
